\documentclass[tikz,border=10pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning,shapes.geometric,fit,backgrounds,calc}

\begin{document}

\definecolor{securegreen}{RGB}{34,139,34}
\definecolor{nscyellow}{RGB}{255,193,7}
\definecolor{nsred}{RGB}{220,53,69}
\definecolor{trustzoneblue}{RGB}{0,123,255}

\begin{tikzpicture}[
    world/.style={draw, rounded corners=8pt, minimum width=7cm, minimum height=10cm, thick},
    component/.style={draw, rounded corners=4pt, minimum width=5.5cm, minimum height=1.2cm, align=center, font=\small},
    nsc/.style={draw, rounded corners=4pt, fill=nscyellow!50, minimum width=2.5cm, minimum height=3cm, align=center, thick},
    arrow/.style={-{Stealth[length=3mm]}, thick, trustzoneblue},
    label/.style={font=\small\bfseries}
]

% Secure World Background
\begin{scope}[on background layer]
    \node[world, fill=securegreen!15] (sworld) at (0,0) {};
\end{scope}
\node[above, label] at (0,5.3) {SECURE WORLD};

% Secure Components
\node[component, fill=securegreen!30] (boot) at (0,4) {Secure Boot\\(MCUboot / BL2)};
\node[component, fill=securegreen!30] (crypto) at (0,2.5) {Crypto Services\\(AES-256-GCM, SHA-256, RNG)};
\node[component, fill=securegreen!30] (gtzc) at (0,1) {GTZC Controller\\(SAU + MPCBB + TZSC)};
\node[component, fill=securegreen!30] (uart) at (0,-0.5) {Debug UART\\(USART1 Secure)};
\node[component, fill=securegreen!30] (storage) at (0,-2) {Protected Storage\\(ITS / PS)};
\node[component, fill=securegreen!30] (attest) at (0,-3.5) {Attestation\\(Device Identity)};

% NSC Gateway
\node[nsc] (nscgate) at (5,0) {NSC\\Gateway\\\\SG Instruction\\Entry Point};

% Non-Secure World Background
\begin{scope}[on background layer]
    \node[world, fill=nsred!15] (nsworld) at (10,0) {};
\end{scope}
\node[above, label] at (10,5.3) {NON-SECURE WORLD};

% Non-Secure Components
\node[component, fill=nsred!25] (app) at (10,4) {Application Logic\\(Command Processing)};
\node[component, fill=nsred!25] (usb) at (10,2.5) {USB CDC Stack\\(USBX Device)};
\node[component, fill=nsred!25] (threadx) at (10,1) {ThreadX RTOS\\(Kernel, Tasks)};
\node[component, fill=nsred!25] (hal) at (10,-0.5) {HAL Drivers\\(USB OTG FS)};
\node[component, fill=nsred!25] (sram) at (10,-2) {NS Memory\\(SRAM2/SRAM3)};

% Arrows from NS to NSC
\draw[arrow] (app.west) -- +(-1.5,0) |- (nscgate.east);
\draw[arrow] (usb.west) -- +(-1,0) |- ([yshift=0.5cm]nscgate.east);

% Arrows from NSC to Secure
\draw[arrow] (nscgate.west) -- +(-1,0) |- (crypto.east);
\draw[arrow] ([yshift=-0.5cm]nscgate.west) -- +(-0.5,0) |- (uart.east);
\draw[arrow] ([yshift=-1cm]nscgate.west) -- +(-0.7,0) |- (storage.east);

% Boot arrow
\draw[arrow, dashed] (boot.east) -- ++(2,0) node[midway, above, font=\tiny] {NS Jump} -- ++(0,-4) -- (app.west);

% Hardware boundary
\draw[thick, dashed, gray] (-4,-5.5) -- (14,-5.5);
\node[below, font=\small\itshape, gray] at (5,-5.5) {Hardware Security Boundary (Cortex-M33 TrustZone)};

% Legend
\node[draw, fill=securegreen!30, minimum width=1cm, minimum height=0.4cm] at (-2,-6.5) {};
\node[right, font=\tiny] at (-1.3,-6.5) {Secure};
\node[draw, fill=nscyellow!50, minimum width=1cm, minimum height=0.4cm] at (2,-6.5) {};
\node[right, font=\tiny] at (2.7,-6.5) {NSC};
\node[draw, fill=nsred!25, minimum width=1cm, minimum height=0.4cm] at (5.5,-6.5) {};
\node[right, font=\tiny] at (6.2,-6.5) {Non-Secure};

\end{tikzpicture}

\end{document}
